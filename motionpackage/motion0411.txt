class FootTrajectoryPlot:
    def __init__(self):
        plt.ion()  # 開啟互動模式
        self.fig, self.ax = plt.subplots()
        # 初始化左右腳軌跡線
        self.lx_line, = self.ax.plot([], [], 'g-', label="Lx")  # 左腳 X (綠色線)
        self.rx_line, = self.ax.plot([], [], 'm-', label="Rx")  # 右腳 X (紫色線)
        # 初始化數據
        self.start_time = time.time()
        self.time = []
        self.lx_values = []
        self.rx_values = []
        # 設定繪圖屬性
        self.ax.set_xlim(0, 20)  # 時間範圍
        self.ax.set_ylim(-5, 5)  # x 位置範圍
        self.ax.set_title("Foot Trajectory X")
        self.ax.set_xlabel("time (s)")
        self.ax.set_ylabel("x (cm)")
        self.ax.legend()

    def update(self, end_point_lx_, end_point_rx_):
        # 更新時間，每次取目前時間減去初始時間，確保是累積時間
        current_time = time.time() - self.start_time
        # 儲存新點
        self.time.append(current_time)
        self.lx_values.append(end_point_lx_)
        self.rx_values.append(end_point_rx_)

        # 更新軌跡線數據
        self.lx_line.set_data(self.time, self.lx_values)
        self.rx_line.set_data(self.time, self.rx_values)

        # 更新繪圖範圍
        self.ax.relim()
        self.ax.autoscale_view()
        plt.pause(0.05)

# 初始化繪圖類別

    def coordinate_transformation(self,py_u, px_u, lpx_, lpy_, rpx_, rpy_, lpz_, rpz_, lpt_, rpt_):
        """座標平移 W to B"""
        step_point_lx_W_ = lpx_ - px_u
        step_point_rx_W_ = rpx_ - px_u
        step_point_ly_W_ = lpy_ - py_u
        step_point_ry_W_ = rpy_ - py_u
        step_point_lz_ = self.pz_ - lpz_
        step_point_rz_ = self.pz_ - rpz_
        step_point_lthta_ = -lpt_
        step_point_rthta_ = -rpt_

        """座標旋轉 W to B"""
        cos_theta = np.cos(-self.theta_)
        sin_theta = np.sin(-self.theta_)
        
        step_point_lx_ = step_point_lx_W_ * cos_theta - step_point_ly_W_ * sin_theta
        step_point_ly_ = step_point_lx_W_ * sin_theta + step_point_ly_W_ * cos_theta
        step_point_rx_ = step_point_rx_W_ * cos_theta - step_point_ry_W_ * sin_theta
        step_point_ry_ = step_point_rx_W_ * sin_theta + step_point_ry_W_ * cos_theta

        end_point_lx_ = step_point_lx_
        end_point_rx_ = step_point_rx_
        end_point_ly_ = step_point_ly_  + self.Length_Pelvis/2
        end_point_ry_ = -step_point_ry_ + self.Length_Pelvis/2
        end_point_lz_ = step_point_lz_ - (self.pz_ - self.Length_Leg)
        end_point_rz_ = step_point_rz_ - (self.pz_ - self.Length_Leg)
        end_point_lthta_ = step_point_lthta_
        end_point_rthta_ = step_point_rthta_
        # print("end_point_rx_",end_point_rx_, "end_point_ry_",end_point_ry_, "end_point_rz_",end_point_rz_, "end_point_lx_",end_point_lx_, "end_point_ly_",end_point_ly_, "end_point_lz_",end_point_lz_, "end_point_rthta_",end_point_rthta_, "end_point_lthta_",end_point_lthta_)
        # print(step_point_ry_-step_point_ly_)
        self.a.append(end_point_rx_)
        self.b.append(step_point_ry_)
        self.c.append(end_point_ry_)
        self.d.append(end_point_lx_)
        self.e.append(step_point_ly_)
        self.f.append(end_point_ly_)
        return end_point_rx_, end_point_ry_, end_point_rz_, end_point_lx_, end_point_ly_, end_point_lz_, end_point_rthta_, end_point_lthta_

    def wComVelocityInit(self, x0, xt, px, t, T):
        return (xt - x0 * np.cosh(t / T) + px * (np.cosh(t / T) - 1)) / (T * np.sinh(t / T))

    def wComPosition(self, x0, vx0, px, t, T):
        return x0 * np.cosh(t / T) + T * vx0 * np.sinh(t / T) - px * (np.cosh(t / T) - 1)

    def wFootPosition(self, start, length, t, T, T_DSP):
        new_T = T * (1 - T_DSP)
        new_t = t - T * T_DSP / 2
        omega = 2 * np.pi / new_T

        if t > 0 and t <= T * T_DSP / 2:
            return start
        elif t > T * T_DSP / 2 and t <= T * (1 - T_DSP / 2):
            return length * (omega * new_t - np.sin(omega * new_t)) / (2 * np.pi) + start
        else:
            return length + start

    def wFootPositionRepeat(self, start, length, t, T, T_DSP):
        new_T = T * (1 - T_DSP)
        new_t = t - T * T_DSP / 2
        omega = 2 * np.pi / new_T

        if t > 0 and t <= T * T_DSP / 2:
            return start
        elif t >= T * T_DSP / 2 and t <= T * (1 - T_DSP / 2):
            return 2 * length * (omega * new_t - np.sin(omega * new_t)) / (2 * np.pi) + start
        else:
            return 2 * length + start

    def wFootPositionZ(self, height, t, T, T_DSP):
        new_T = T * (1 - T_DSP)
        new_t = t - T * T_DSP / 2
        omega = 2 * np.pi / new_T

        if T * T_DSP / 2 < t < T * (1 - (T_DSP / 2)):
            return 0.5 * height * (1 - np.cos(omega * new_t))
        else:
            return 0

    def wFootTheta(self, theta, reverse, t, T, T_DSP):
        new_T = T * (1 - T_DSP)
        new_t = t - T * T_DSP / 2
        omega = 2 * np.pi / new_T
        
        if t > 0 and t <= T * T_DSP / 2:
            return theta if reverse else 0
        elif T * T_DSP / 2 < t <= T * (1 - T_DSP / 2):
            if reverse:
                return 0.5 * theta * (1 - np.cos(0.5 * omega * (new_t - new_T)))
            else:
                return 0.5 * theta * (1 - np.cos(0.5 * omega * new_t))
        else:
            return 0 if reverse else theta

    def serial_walk(self):

        self.baudrate = 115200

    def angle_to_motor_scale(self, theta, angle_range=2 * math.pi, scale_range=4096):
        """
        將角度值轉換為馬達刻度值
        """
        return int(2048 + theta * (scale_range / angle_range))

    def calculate_motor_speed(self):
        """
        計算馬達速度（刻度/秒）
        """
        # speed_trans =  32767/1023
        all_speed_gain = 1//2
        # self.diff_matrix = np.zeros((10,22))
        
        for i in range(10, 22):
            self.difference = math.fabs(self.motor_values[i] - self.prev_motor_values[i])
            if self.difference > 0:
                speed = (self.difference/(4096)) * (1000/self.motion_delay)*60/0.229
            else:
                speed = 0
            self.motor_speeds[i] = int(speed)
            # print("motor_speeds",self.motor_speeds[i])
            self.prev_motor_values[i] = self.motor_values[i]  # 更新上一刻度

        return self.motor_speeds


    def init(self, end_point_rx_, end_point_ry_, end_point_rz_, end_point_lx_, end_point_ly_, end_point_lz_, end_point_rthta_, end_point_lthta_):
        self.l1 = 12.5
        self.l2 = 12.5
        self.RX_2 = end_point_rx_ * end_point_rx_
        self.RY_2 = end_point_ry_ * end_point_ry_
        self.RZ_2 = end_point_rz_ * end_point_rz_
        self.LX_2 = end_point_lx_ * end_point_lx_
        self.LY_2 = end_point_ly_ * end_point_ly_
        self.LZ_2 = end_point_lz_ * end_point_lz_
        self.l1_2 = self.l1 * self.l1
        self.l2_2 = self.l2 * self.l2
        self.l1_l2 = self.l1 + self.l2
        self.R_Lyz = np.sqrt(self.RY_2 + self.RZ_2)
        self.L_Lyz = np.sqrt(self.LY_2 + self.LZ_2)
        self.R_Lxyz = np.sqrt(self.RX_2 + self.RY_2 + self.RZ_2)
        self.L_Lxyz = np.sqrt(self.LX_2 + self.LY_2 + self.LZ_2)

    def ik(self, end_point_rx_=0, end_point_ry_=0, end_point_rz_=23.5, 
           end_point_lx_=0, end_point_ly_=0, end_point_lz_=23.5, 
           end_point_rthta_=0, end_point_lthta_=0):
        self.init(end_point_rx_, end_point_ry_, end_point_rz_, 
                  end_point_lx_, end_point_ly_, end_point_lz_, 
                  end_point_rthta_, end_point_lthta_)

        # 限制距離防止超出可動範圍
        if self.R_Lxyz > self.l1_l2:
            self.R_Lxyz = self.l1_l2
        if self.L_Lxyz > self.l1_l2:
            self.L_Lxyz = self.l1_l2

        self.RL_2 = self.R_Lxyz * self.R_Lxyz
        self.LL_2 = self.L_Lxyz * self.L_Lxyz

        # 左腳角度計算
        self.Thta[10] = 0
        self.Thta[11] = math.atan2(end_point_lz_, end_point_ly_) - np.pi/2 if end_point_ly_ != 0 else 0

        if end_point_lx_ == 0:
            self.Thta[12] = -math.acos((self.l1_2 + self.LL_2 - self.l2_2) / (2 * self.l1 * self.L_Lxyz))
        elif end_point_lx_ > 0:
            self.Thta[12] = -math.acos((self.l1_2 + self.LL_2 - self.l2_2) / (2 * self.l1 * self.L_Lxyz)) - math.atan2(end_point_lx_, self.L_Lyz)
        else:
            self.Thta[12] = np.pi/2 - math.acos((self.l1_2 + self.LL_2 - self.l2_2) / (2 * self.l1 * self.L_Lxyz)) - math.atan2(self.L_Lyz, -end_point_lx_)

        self.Thta[13] = np.pi - math.acos((self.l1_2 + self.l2_2 - self.LL_2) / (2 * self.l1 * self.l2))
        self.Thta[14] = -(self.Thta[12] + self.Thta[13])
        self.Thta[15] = self.Thta[11]

        # 右腳角度計算
        self.Thta[16] = 0
        self.Thta[17] = math.atan2(end_point_rz_, end_point_ry_) - np.pi/2 if end_point_ry_ != 0 else 0

        if end_point_rx_ == 0:
            self.Thta[18] = -math.acos((self.l1_2 + self.RL_2 - self.l2_2) / (2 * self.l1 * self.R_Lxyz))
        elif end_point_rx_ > 0:
            self.Thta[18] = -math.acos((self.l1_2 + self.RL_2 - self.l2_2) / (2 * self.l1 * self.R_Lxyz)) - math.atan2(end_point_rx_, self.R_Lyz)
        else:
            self.Thta[18] = np.pi/2 - math.acos((self.l1_2 + self.RL_2 - self.l2_2) / (2 * self.l1 * self.R_Lxyz)) - math.atan2(self.R_Lyz, -end_point_rx_)

        self.Thta[19] = -(np.pi - math.acos((self.l1_2 + self.l2_2 - self.RL_2) / (2 * self.l1 * self.l2)))
        self.Thta[20] = -(self.Thta[18] + self.Thta[19])-(np.pi/2)
        self.Thta[21] =  -(self.Thta[17])
        for i in range(1,11):
            self.idex[i] = i
        # 計算馬達刻度
        for i in range(10, 22):
            self.idex[i] = i
            self.motor_values[i] = int(self.angle_to_motor_scale(self.Thta[i]))
        # 計算馬達速度
        self.motion_delay = 30#self.walk.period_t_ / self.walk.sample_time_
        # print("motion_delay", self.motion_delay)
        self.calculate_motor_speed()
        self.WalkingMotorFunction()
        # print(f"LF (角度): {self.Thta[10:16]}")
        # print(f"LF (馬達刻度): {self.motor_values[10:16]}")
        # print(f"RF (角度): {self.Thta[20:22]}")
        # print(f"RF (馬達刻度): {self.motor_values[16:22]}")
        plot_foot_trajectory(end_point_lx_, end_point_rx_)

        print(f"walking_mode: {self.walking_mode}")
        self.walking_state = msg.walking_state
        print(f"walking_state: {self.walking_state}")
        self.sensor_mode = msg.sensor_mode
        print(f"sensor_mode: {self.sensor_mode}")

        # 建立狀態矩陣（1x7）
        self.state_vector = np.array([
            self.x,
            self.y,
            self.z,
            self.theta,
            self.walking_mode,
            self.walking_state,
            self.sensor_mode
        ])

        # 建立 serial 封包
        HEADER = 0xAA
        END = 0x55

        # 將資料打包成 bytes（4個 float + 3個 uint8）
        payload = struct.pack(
            '<ffffBBB',
            self.x,
            self.y,
            self.z,
            self.theta,
            int(self.walking_mode),
            int(self.walking_state),
            int(self.sensor_mode)
        )

        # 封裝成完整封包
        self.serial_packet = bytearray()
        self.serial_packet.append(HEADER)
        self.serial_packet.extend(payload)
        self.serial_packet.append(END)
        print(f"Serial Packet: {self.serial_packet}")

    def create_sensorsetpackage(self, parameters, setopt):
        sensorsetpackage = [0] * 12
        sensorsetpackage[0] = 0x53
        sensorsetpackage[1] = 0x54
        sensorsetpackage[2] = 0xF6

        for i in range(3):
            if parameters[i] < 0:
                parameters[i] = ~(parameters[i]) + 1
                sensorsetpackage[3 + i * 2] = ((parameters[i] >> 8) & 0xFF) | 0x80
                sensorsetpackage[4 + i * 2] = parameters[i] & 0xFF
                parameters[i] = ~(parameters[i] - 1)
            else:
                sensorsetpackage[3 + i * 2] = (parameters[i] >> 8) & 0xFF
                sensorsetpackage[4 + i * 2] = parameters[i] & 0xFF

        sensorsetpackage[9] = setopt
        sensorsetpackage[10] = 0  # Reserve
        sensorsetpackage[11] = 0x45

        return sensorsetpackage

    def Sensor_Data_Process(self, buf):
        Sensor_Data_tmp = [0] * 11
        IMU_Value = [0] * 3
        Sensor_Data_Count = 3

        for i in range(3):
            # Combine the high 8 bits and the low 8 bits
            Sensor_Data_tmp[i] = (buf[Sensor_Data_Count] << 8) | buf[Sensor_Data_Count+1]
            # Check if the data is negative
            if Sensor_Data_tmp[i] & 0x8000:  # If the sign bit is 1
                # Convert the data from two's complement to its actual integer value,
                # and then convert it to a floating-point number
                IMU_Value[i] = ((~(Sensor_Data_tmp[i] & 0x7FFF) + 1) / 100.0)
            else:  # If the data is positive
                # Convert the data to a floating-point number
                IMU_Value[i] = Sensor_Data_tmp[i] / 100.0
            Sensor_Data_Count += 2  # Update Sensor_Data_Count
        # Create a new SensorPackage message
        msg = SensorPackage()
        msg.imudata = IMU_Value  # Set the IMUData attribute to the IMU_Value array
        # Publish the message
        self.sensor_data_pub.publish(msg)

    def motor_setPID(self, msg):
        motorID = msg.motori

        motor_P_L = msg.motor_p & 0xFF
        motor_P_H = ((msg.motor_p)>>8) & 0xFF
        motor_I_L = msg.motor_i & 0xFF
        motor_I_H = ((msg.motor_i)>>8) & 0xFF
        motor_D_L = msg.motor_d & 0xFF
        motor_D_H = ((msg.motor_d)>>8) & 0xFF

        print(f"ID:{motorID},set_P:{msg.motor_p},set_I:{msg.motor_i},set_D:{msg.motor_d}")
        print(f"P:{msg.pflag},I:{msg.iflag},D:{msg.dflag}")

        PIDpackage = [0]*14

        PIDpackage[0] = 0xFF
        PIDpackage[1] = 0xFF
        PIDpackage[2] = 0xFD
        PIDpackage[3] = 0x00
        PIDpackage[4] = motorID
        PIDpackage[5] = 0x07
        PIDpackage[6] = 0
        PIDpackage[7] = 0x3

        if msg.Pflag:
            PIDpackage[8]  = 0x54
            PIDpackage[10] = motor_P_L
            PIDpackage[11] = motor_P_H
        elif msg.Iflag:
            PIDpackage[8]  = 0x52
            PIDpackage[10] = motor_I_L
            PIDpackage[11] = motor_I_H
        elif msg.Dflag:
            PIDpackage[8]  = 0x50
            PIDpackage[10] = motor_D_L
            PIDpackage[11] = motor_D_H

        PIDpackage[9] = 0

        blk_size = 5 + PIDpackage[5]
        crc_value = self.update_crc(0, PIDpackage, blk_size)

        PIDpackage[12] = (crc_value << 8) >> 8
        PIDpackage[13] = crc_value >> 8

        for cnt in range(14):
            self.serial_motor.write(PIDpackage[cnt].to_bytes(1, byteorder='big'))

    def _write_motion_list(self, fout, motionlist):
        for j, value in enumerate(motionlist):
            prefix = "A" if j % 2 == 0 else "D"
            fout.write(f"{prefix}{(j // 2) + 1} = {value}")
            if j != len(motionlist) - 1:
                fout.write("|")
        fout.write("\n")

    def _write_motor_data(self, fout, motordata):
        for j, value in enumerate(motordata):
            fout.write(f"M{j + 1} = {value}")
            if j != len(motordata) - 1:
                fout.write("|")
        fout.write("\n")

                    response.relativedata = [item for sublist in relativedata_list for item in sublist if -32768 <= item <= 32767]
                elif current_state in [3, 4] and line.startswith("M"):
                    m_values = [int(m_value.split('=')[1].strip()) for m_value in line.split('|')]

                    response.absolutedata = [item for sublist in absolutedata_list for item in sublist if -32768 <= item <= 32767]

                    ad_values = [int(ad_value.split('=')[1].strip()) for ad_value in line.split('|')]
                    motion_list.append(ad_values)
                    response.motionlist = [item for sublist in motion_list for item in sublist if -32768 <= item <= 32767]
